        // --- State Management ---
        const STORAGE_KEY = 'kanban_board_state';
        
        let state = {
            columns: [
                { id: 'backlog', title: 'Backlog', cards: [] },
                { id: 'todo', title: 'To Do', cards: [] },
                { id: 'inprogress', title: 'In Progress', cards: [] },
                { id: 'done', title: 'Done', cards: [] }
            ]
        };

        // --- DOM Elements ---
        const boardEl = document.getElementById('board');

        const priorityFilter = document.getElementById('priorityFilter');
        const dropZone = document.getElementById('dropZone');

        // --- Initialization ---
        function init() {
            const savedState = localStorage.getItem(STORAGE_KEY);
            if (savedState) {
                state = JSON.parse(savedState);
            } else {
                // Seed data
                state.columns[0].cards.push(createCard('Research project requirements', 'high', '2024-01-15'));
                state.columns[0].cards.push(createCard('Setup development environment', 'medium', null));
                state.columns[1].cards.push(createCard('Design database schema', 'high', '2024-01-20'));
                state.columns[1].cards.push(createCard('Create API endpoints', 'medium', null));
                state.columns[2].cards.push(createCard('Implement user authentication', 'high', '2024-01-25'));
                state.columns[3].cards.push(createCard('Deploy to production', 'low', '2024-02-01'));
            }
            renderBoard();
            updateStats();
            setupEventListeners();
        }

        function createCard(title, priority, dueDate) {
            return {
                id: 'card_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9),
                title: title,
                description: '',
                priority: priority || 'medium',
                dueDate: dueDate || null,
                isDone: false
            };
        }

        // --- Rendering ---

            // Clear existing columns
            boardEl.innerHTML = '<div class="drop-zone" id="dropZone"></div>';
